git checkout prod — перемикання на гілку готового продукту для внесення основних змін.

python -m poetry add psycopg2-binary — додавання драйвера (бібліотеки) для того, щоб Python міг працювати з базою даних PostgreSQL.

docker compose up --build — команда, яка збирає образи на основі Dockerfile та запускає обидва контейнери (FastAPI та Postgres) одночасно.

docker exec -it fastapi-app-1 pip list — команда для «входу» в працюючий контейнер та виведення списку всіх встановлених у ньому бібліотек.

git add . — підготовка всіх нових файлів (Dockerfile, docker-compose.yml, .env тощо) до збереження.

git commit -m "..." — фіксація змін в історії гілки.

git push origin prod — відправка коду з Docker-налаштуваннями на GitHub у гілку prod.

git checkout dev — перехід на гілку розробки для підготовки звіту.

git merge prod — злиття (копіювання) всього нового коду та налаштувань з гілки prod у гілку dev.

git log > git_logs.txt — створення (або оновлення) текстового файлу з історією всіх комітів.
